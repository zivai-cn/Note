1. 按键状态结构体
```c
// 按钮事件类型
typedef enum {
    BTN_PRESS_DOWN = 0,     // 按钮按下
    BTN_PRESS_UP,           // 按钮释放
    BTN_PRESS_REPEAT,       // 检测到重复按下
    BTN_SINGLE_CLICK,       // 单次点击完成
    BTN_DOUBLE_CLICK,       // 双击完成
    BTN_LONG_PRESS_START,   // 长按开始
    BTN_LONG_PRESS_HOLD,    // 长按持续
    BTN_EVENT_COUNT,        // 总事件数
    BTN_NONE_PRESS          // 无事件
} ButtonEvent
// 按钮状态机状态
typedef enum {
BTN_STATE_IDLE = 0,     // 空闲状态
BTN_STATE_PRESS，        // 按下状态
BTN_STATE_RELEASE，      // 释放状态（等待超时）
BTN_STATE_REPEAT，       // 重复按下状态
BTN_STATE_LONG_HOLD     // 长按保持状态
} ButtonState；
// 按钮结构
struct _Button {
  uint16_t ticks；                     // 计时器计数值
  uint8_t  repeat ： 4；                // 重复计数器（0 - 15）
  uint8_t  event ： 4；                 // 当前事件（0 - 15）
  uint8_t  state ： 3；                 // 状态机状态（0 - 7）
  uint8_t  debounce_cnt ： 3；          // 去抖计数器（0 - 7）
  uint8_t  active_level ： 1；          // 活动 GPIO 级别（0 或 1）
  uint8_t  button_level ： 1；          // 当前按钮级别
  uint8_t  button_id；                 // 按钮标识符
  uint8_t  (*hal_button_level)(uint8_t button_id)；  // HAL 函数用于读取 GPIO
  BtnCallback cb[BTN_EVENT_COUNT]；    // 回调函数数组
  void*    user_data；                 // 传递给回调函数的用户上下文指针
  Button* next；                       // 链表中下一个按钮指针
}
};

```
1. 核心功能函数
`button_handler`,通过状态机的轮询，判别事件的处理。
```c
/**
  * @brief  Button driver core function, driver state machine
  * @param  handle: the button handle struct
  * @retval None
  */
static void button_handler(Button* handle)
{	
    //1.读取GPIO口的电平 
    uint8_t read_gpio_level = button_read_level(handle); 
    // Increment ticks counter when not in idle state (with saturation)
    //2.判断是否处在空闲状态，BTN_STATE_IDLE是空闲态位，因为作为枚举量的首位，所以大于就是判断非BTN_STATE_IDLE态。确定是非空闲态之后，就开始对ticks进行轮询自增，为后续的长按等状态的判定做准备。
    if (handle->state > BTN_STATE_IDLE) {
        if (handle->ticks < UINT16_MAX) {
            handle->ticks++;
        }
    }
    //3.消抖代码，button_level是上一次的按键状态，如果本次检测到的状态和上次不同，对debounce_cnt自增，DEBOUNCE_TICKS是最大系数，如果多次检测的结果都不同，认为电平发生了变化，同时计数器归零。
    if (read_gpio_level != handle->button_level) {
        // Continue reading same new level for debounce
        if (++(handle->debounce_cnt) >= DEBOUNCE_TICKS) {
            handle->button_level = read_gpio_level;
            handle->debounce_cnt = 0;
        }
    } else {
        // Level not changed, reset counter
        handle->debounce_cnt = 0;
    }
    //4.状态机判断，判断状态，执行相应的功能。
    /* State machine */
    switch (handle->state) {
    //(1)按键无工作。active_level活动状态，如果按键是active_level态，event设为BTN_PRESS_DOWN，通知应用层，计时清零，repeat重复次数记1次，state改为BTN_STATE_PRESS。
    case BTN_STATE_IDLE:
        if (handle->button_level == handle->active_level) {
            // Button press detected
            handle->event = (uint8_t)BTN_PRESS_DOWN;
            EVENT_CB(BTN_PRESS_DOWN);
            handle->ticks = 0;
            handle->repeat = 1;
            handle->state = BTN_STATE_PRESS;
        } else {
            handle->event = (uint8_t)repeat;
        }
        break;
    //(2)按键按下中。若按键不再是active_level，event设为BTN_PRESS_UP，通知应用层，计时清零，state改为BTN_STATE_RELEASE。若按键处于active_level，且时间更长，超过LONG_TICKS，判断为长按，event改为BTN_LONG_PRESS_START，state改为BTN_STATE_LONG_HOLD。
    case BTN_STATE_PRESS:
        if (handle->button_level != handle->active_level) {
            // Button released
            handle->event = (uint8_t)BTN_PRESS_UP;
            EVENT_CB(BTN_PRESS_UP);
            handle->ticks = 0;
            handle->state = BTN_STATE_RELEASE;
        } else if (handle->ticks > LONG_TICKS) {
            // Long press detected
            handle->event = (uint8_t)BTN_LONG_PRESS_START;
            EVENT_CB(BTN_LONG_PRESS_START);
            handle->state = BTN_STATE_LONG_HOLD;
        }
        break;
    //(3)按键已经松开，需要判断是否为多次点击的状态。若判定为按下，event改为BTN_PRESS_DOWN，repeat重复次数自增，event再改为BTN_PRESS_REPEAT，通知应用层，state改为BTN_STATE_REPEAT。若ticks < SHORT_TICKS，无操作，ticks > SHORT_TICKS，对
    case BTN_STATE_RELEASE:
        if (handle->button_level == handle->active_level) {
            // Button pressed again
            handle->event = (uint8_t)BTN_PRESS_DOWN;
            EVENT_CB(BTN_PRESS_DOWN);
            if (handle->repeat < PRESS_REPEAT_MAX_NUM) {
                handle->repeat++;
            }
            handle->event = (uint8_t)BTN_PRESS_REPEAT;
            EVENT_CB(BTN_PRESS_REPEAT);
            handle->ticks = 0;
            handle->state = BTN_STATE_REPEAT;
        } else if (handle->ticks > SHORT_TICKS) {
            // Timeout reached, determine click type
            if (handle->repeat == 1) {
                handle->event = (uint8_t)BTN_SINGLE_CLICK;
                EVENT_CB(BTN_SINGLE_CLICK);
            } else if (handle->repeat == 2) {
                handle->event = (uint8_t)BTN_DOUBLE_CLICK;
                EVENT_CB(BTN_DOUBLE_CLICK);
            }
            handle->state = BTN_STATE_IDLE;
        }
        break;
    case BTN_STATE_REPEAT:
        if (handle->button_level != handle->active_level) {
            // Button released
            handle->event = (uint8_t)BTN_PRESS_UP;
            EVENT_CB(BTN_PRESS_UP);
            if (handle->ticks < SHORT_TICKS) {
                handle->ticks = 0;
                handle->state = BTN_STATE_RELEASE;  // Continue waiting for more presses
            } else {
                handle->state = BTN_STATE_IDLE;  // End of sequence
            }
        } else if (handle->ticks > SHORT_TICKS) {
            // Held down too long, treat as normal press
            handle->ticks = 0;      // reset for fresh long-press timing
            handle->repeat = 0;     // clear repeat count for new press cycle
            handle->state = BTN_STATE_PRESS;
        }
        break;  
    case BTN_STATE_LONG_HOLD:
        if (handle->button_level == handle->active_level) {
            // Continue holding
            handle->event = (uint8_t)BTN_LONG_PRESS_HOLD;
            EVENT_CB(BTN_LONG_PRESS_HOLD);
        } else {
            // Released from long press
            handle->event = (uint8_t)BTN_PRESS_UP;
            EVENT_CB(BTN_PRESS_UP);
            handle->state = BTN_STATE_IDLE;
        }
        break;
    default:
        // Invalid state, reset to idle
        handle->state = BTN_STATE_IDLE;
        break;
    }
}
```
